\chapter{Notation}\label{app:notation}

This appendix lists the conventions used throughout the book. They are defined as macros in \texttt{notation.tex}; keep the two in sync.

\section*{Sets and numbers}
\begin{tabular}{@{}lp{0.62\textwidth}@{}}
\toprule
Notation & Meaning \\
\midrule
$\N, \Z, \Q, \R, \C$ & naturals $\set{1, 2, \ldots}$, integers, rationals, reals, complex numbers \\
$\R^{+}, \R^{-}$ & positive reals, negative reals \\
$\powerset(\Omega)$ & power set: the set of all subsets of $\Omega$ \\
$\abs{x}, \norm{\vect{x}}$ & absolute value, norm \\
$\inner{\vect{x}, \vect{y}}$ & inner product \\
\bottomrule
\end{tabular}

\section*{Linear algebra}
\begin{tabular}{@{}lp{0.62\textwidth}@{}}
\toprule
Notation & Meaning \\
\midrule
$\vect{x}$ & vector (bold, column vector by default) \\
$\mat{A}$ & matrix (bold upper case) \\
$\mat{A}\T$ & transpose \\
$\tr \mat{A}, \rank \mat{A}, \diag(\cdot)$ & trace, rank, diagonal matrix \\
\bottomrule
\end{tabular}

\section*{Probability and statistics}
\begin{tabular}{@{}lp{0.62\textwidth}@{}}
\toprule
Notation & Meaning \\
\midrule
$\Prob(A), \Prob(A \given B)$ & probability, conditional probability \\
$\E(X), \Var(X), \Cov(X, Y)$ & expectation (also written $\E[X]$ or $\E X$), variance, covariance \\
$F_X$ & cumulative distribution function of $X$ \\
$\ind_A$ & indicator of the event $A$ \\
$X \indep Y$ & $X$ and $Y$ are independent \\
$X_1, \ldots, X_n \iid F$ & independent and identically distributed from $F$ \\
$X \eqd Y$ & equal in distribution \\
$\convp, \convd, \convas$ & convergence in probability, in distribution, almost surely \\
$\Normal(\mu, \sigma^2)$ & normal distribution, parametrised by \emph{variance} \\
$\Expo(\lambda)$ & exponential distribution, \emph{rate} $\lambda$ (mean $1/\lambda$) \\
$\Bern(p)$, $\Bin(n, p)$, $\Geom(p)$ & Bernoulli, binomial, geometric (number of trials until the first success) \\
$\NegBin(k, p)$ & negative binomial (number of trials until the $k$th success) \\
$\Pois(\lambda)$, $\Hyp(n, r, N)$ & Poisson, hypergeometric \\
$\Unif[a, b]$ & uniform distribution on $[a, b]$ \\
$\GammaDist(\alpha, \lambda)$ & gamma distribution, shape $\alpha$, \emph{rate} $\lambda$ (sometimes written with scale $\beta = 1/\lambda$) \\
$\ChiSq{n}$, $\BetaDist(\alpha, \beta)$ & chi-squared with $n$ degrees of freedom; beta distribution \\
$\phi$, $\Phi$ & standard normal pdf and CDF \\
\bottomrule
\end{tabular}
